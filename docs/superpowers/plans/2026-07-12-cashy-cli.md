# Cashy CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A thin `cashy` CLI over the existing `/api/*` HTTP API: token login via a browser page + loopback server, balance updates (single + bulk CSV/JSON import), and dashboard status display.

**Architecture:** Server side, better-auth is upgraded to 1.6.23 and the `@better-auth/api-key` plugin is added with `enableSessionForAPIKeys: true`, so a valid `x-api-key` header acts as a session and the existing `sessionMiddleware`/`workspaceMiddleware` work unchanged (workspace falls back to the user's first workspace). A new session-protected HTML page `/cli/auth` mints a key in the browser and POSTs it to a loopback server run by `cashy auth login`. The CLI itself lives in `src/cli/` (commander), with all pure logic (amounts, CSV, matrix validation, account resolution) in dependency-free, unit-tested modules.

**Tech Stack:** Bun (runtime + `bun test` + `Bun.serve` loopback), commander, better-auth 1.6.23 + `@better-auth/api-key`, drizzle (schema push), existing Hono API (unchanged except auth plugin).

**Spec:** `docs/superpowers/specs/2026-07-12-cashy-cli-design.md`

## Global Constraints

- Use Bun for everything: `bun add`, `bun test`, `bun <file>` (per CLAUDE.md). No npm/yarn/pnpm.
- New runtime dependencies allowed: `commander`, `better-auth@^1.6.23` (upgrade), `@better-auth/api-key@^1.6.23`. Nothing else (no csv/chalk/ora).
- CLI entry: `bun run cashy …` via a package.json script; no build step, no npm publishing.
- All CLI requests send the token as the `x-api-key` header.
- Config resolution order (first wins): flags `--api-url`/`--token` → env `CASHY_API_URL`/`CASHY_TOKEN` → `~/.config/cashy/config.json` (written mode 0600).
- Amounts: CLI accepts major units with ≤ 2 decimals (`1234.56`), converts to integer minor units (×100). API field is `amountMinorRaw`.
- Exit codes: `0` success, `1` network/unexpected, `2` validation, `3` unauthorized/forbidden, `4` not found.
- Import: validate everything first (report ALL issues, exit 2, write nothing), then one `POST /api/balance-snapshots/bulk` per date, oldest first. Empty cells skipped. Not atomic across dates — on failure report applied vs. not-applied dates.
- Local dev server runs at `http://localhost:3050` (from `.env`). The CLI's built-in default API URL is `http://localhost:3050`.
- Match existing code style: path alias `@/` (see `tsconfig.json`), 2-space indent, named exports.

```ts
import { homedir } from "node:os";
import { mkdir, unlink, chmod } from "node:fs/promises";
import path from "node:path";

export type CliConfig = { apiUrl: string; token: string };

export const DEFAULT_API_URL = "http://localhost:3050";

export const CONFIG_PATH = path.join(
  homedir(),
  ".config",
  "cashy",
  "config.json",
);

export function resolveConfig(
  flags: { apiUrl?: string; token?: string },
  env: Record<string, string | undefined>,
  fileConfig: Partial<CliConfig>,
): { apiUrl: string; token?: string } {
  return {
    apiUrl:
      flags.apiUrl ?? env.CASHY_API_URL ?? fileConfig.apiUrl ?? DEFAULT_API_URL,
    token: flags.token ?? env.CASHY_TOKEN ?? fileConfig.token,
  };
}

export async function readConfigFile(): Promise<Partial<CliConfig>> {
  const file = Bun.file(CONFIG_PATH);
  if (!(await file.exists())) return {};
  try {
    return (await file.json()) as Partial<CliConfig>;
  } catch {
    return {};
  }
}

export async function writeConfigFile(config: CliConfig): Promise<void> {
  await mkdir(path.dirname(CONFIG_PATH), { recursive: true });
  await Bun.write(CONFIG_PATH, JSON.stringify(config, null, 2) + "\n");
  await chmod(CONFIG_PATH, 0o600);
}

export async function deleteConfigFile(): Promise<void> {
  try {
    await unlink(CONFIG_PATH);
  } catch {
    // already gone
  }
}
```

```ts
export class ApiError extends Error {
  constructor(
    public code: string,
    message: string,
    public status: number,
    public details?: unknown,
  ) {
    super(message);
  }
}

export type ApiAccount = { id: string; name: string; type: string };

export type BulkInput = {
  effectiveDate: string;
  snapshots: { financialAccountId: string; amountMinorRaw: number }[];
};

export type SessionInfo = {
  user: { id: string; name: string; email: string };
  workspaces: { id: string; name: string }[];
  activeWorkspaceId: string;
};

export type DashboardSummary = {
  currentBalance: number;
  monthlyIncomeTotal: number;
  monthlyExpenseTotal: number;
  monthlySpendingTotal: number;
  monthlySpendingExcludingTracked: number;
  lastSnapshotDate: string | null;
  recurringIncomeTotal: number;
  recurringExpenseTotal: number;
  budgetItemsTotal: number;
};

export type DashboardResponse = {
  summary: DashboardSummary;
  isCurrentMonth: boolean;
};

export class CashyClient {
  constructor(
    private apiUrl: string,
    private token?: string,
  ) {}

  private async request<T>(
    method: string,
    apiPath: string,
    body?: unknown,
  ): Promise<T> {
    if (!this.token) {
      throw new ApiError(
        "UNAUTHORIZED",
        "No token configured. Run: bun run cashy auth login",
        401,
      );
    }
    let res: Response;
    try {
      res = await fetch(new URL(apiPath, this.apiUrl), {
        method,
        headers: {
          "x-api-key": this.token,
          ...(body !== undefined
            ? { "content-type": "application/json" }
            : {}),
        },
        body: body !== undefined ? JSON.stringify(body) : undefined,
      });
    } catch (err) {
      throw new ApiError(
        "NETWORK",
        `Cannot reach ${this.apiUrl} — is the server running? (${String(err)})`,
        0,
      );
    }
    if (!res.ok) {
      let code = "INTERNAL";
      let message = `HTTP ${res.status}`;
      let details: unknown;
      try {
        const parsed = (await res.json()) as {
          error?: { code?: string; message?: string; details?: unknown };
          message?: string;
        };
        code = parsed.error?.code ?? code;
        message = parsed.error?.message ?? parsed.message ?? message;
        details = parsed.error?.details;
      } catch {
        // non-JSON error body
      }
      throw new ApiError(code, message, res.status, details);
    }
    return (await res.json()) as T;
  }

  getSession(): Promise<SessionInfo> {
    return this.request("GET", "/api/session");
  }

  async listAccounts(): Promise<ApiAccount[]> {
    const data = await this.request<{ accounts: ApiAccount[] }>(
      "GET",
      "/api/accounts",
    );
    return data.accounts;
  }

  bulkSnapshots(input: BulkInput): Promise<unknown> {
    return this.request("POST", "/api/balance-snapshots/bulk", input);
  }

  getDashboard(month?: string): Promise<DashboardResponse> {
    const q = month ? `?month=${encodeURIComponent(month)}` : "";
    return this.request("GET", `/api/dashboard${q}`);
  }
}

export function exitCodeForError(err: unknown): number {
  if (err instanceof ApiError) {
    if (err.status === 401 || err.status === 403) return 3;
    if (err.status === 404) return 4;
    if (err.status === 400 || err.code === "VALIDATION_ERROR") return 2;
  }
  return 1;
}
```

```ts
const AMOUNT_RE = /^-?\d+(\.\d{1,2})?$/;

/** Parse a major-unit amount string ("1234.56") into integer minor units. */
export function parseMajorAmount(input: string): number {
  const trimmed = input.trim();
  if (!/^-?\d+(\.\d+)?$/.test(trimmed)) {
    throw new Error(`"${input}" is not a valid amount`);
  }
  if (!AMOUNT_RE.test(trimmed)) {
    throw new Error(`"${input}" has more than 2 decimal places`);
  }
  const negative = trimmed.startsWith("-");
  const [wholePart, fracPart = ""] = (negative ? trimmed.slice(1) : trimmed)
    .split(".") as [string, string?];
  const minor =
    Number(wholePart) * 100 + Number((fracPart ?? "").padEnd(2, "0") || "0");
  if (!Number.isSafeInteger(minor)) {
    throw new Error(`"${input}" is too large`);
  }
  return negative ? -minor : minor;
}

/** Format integer minor units as "1 234.56" (space thousands grouping). */
export function formatMinor(minor: number): string {
  const negative = minor < 0;
  const abs = Math.abs(minor);
  const whole = Math.floor(abs / 100).toString();
  const frac = (abs % 100).toString().padStart(2, "0");
  const grouped = whole.replace(/\B(?=(\d{3})+(?!\d))/g, " ");
  return `${negative ? "-" : ""}${grouped}.${frac}`;
}
```

```ts
/** Minimal CSV parser: quoted fields, "" escapes, LF/CRLF, BOM, skips empty lines. */
export function parseCsv(text: string): string[][] {
  const src = text.startsWith("﻿") ? text.slice(1) : text;
  const rows: string[][] = [];
  let row: string[] = [];
  let cell = "";
  let quoted = false;
  let cellWasQuoted = false;

  const pushCell = () => {
    row.push(cellWasQuoted ? cell : cell.trim());
    cell = "";
    cellWasQuoted = false;
  };
  const pushRow = () => {
    pushCell();
    if (!(row.length === 1 && row[0] === "")) rows.push(row);
    row = [];
  };

  for (let i = 0; i < src.length; i++) {
    const ch = src[i];
    if (quoted) {
      if (ch === '"') {
        if (src[i + 1] === '"') {
          cell += '"';
          i++;
        } else {
          quoted = false;
        }
      } else {
        cell += ch;
      }
    } else if (ch === '"' && cell.trim() === "") {
      quoted = true;
      cellWasQuoted = true;
      cell = "";
    } else if (ch === ",") {
      pushCell();
    } else if (ch === "\n") {
      pushRow();
    } else if (ch === "\r") {
      // swallow (CRLF handled by \n)
    } else {
      cell += ch;
    }
  }
  if (cell !== "" || cellWasQuoted || row.length > 0) pushRow();
  return rows;
}
```

```ts
export function resolveAccountRef(
  ref: string,
  accounts: ApiAccount[],

  const byId = accounts.find((a) => a.id === ref);
```

```ts
import path from "node:path";
import { isDateString } from "@/lib/date";
import type { ApiAccount } from "../client";
import { parseCsv } from "./csv";
import { parseMajorAmount } from "./amounts";
import { resolveAccountRef, accountListHint } from "./accounts";

export type MatrixCell = { accountRef: string; rawAmount: string };
export type MatrixRow = { date: string; cells: MatrixCell[] };

export type DatePlan = {
  date: string;
  snapshots: {
    financialAccountId: string;
    accountName: string;
    amountMinorRaw: number;
  }[];
};

export function parseImportFile(
  filename: string,
  content: string,
): { rows: MatrixRow[]; issues: string[] } {
  const ext = path.extname(filename).toLowerCase();
  if (ext === ".csv") return parseCsvMatrix(content);
  if (ext === ".json") return parseJsonMatrix(content);
  return {
    rows: [],
    issues: [`Unsupported file type: ${ext} (use .csv or .json)`],
  };
}

function parseCsvMatrix(content: string): {
  rows: MatrixRow[];
  issues: string[];
} {
  const issues: string[] = [];
  const grid = parseCsv(content);
  if (grid.length === 0) return { rows: [], issues: ["File is empty"] };

  const header = grid[0]!;
  if ((header[0] ?? "").toLowerCase() !== "date") {
    return {
      rows: [],
      issues: [`CSV header must start with a "date" column (got "${header[0] ?? ""}")`],
    };
  }
  const accountRefs = header.slice(1);
  const seenRefs = new Set<string>();
  for (const ref of accountRefs) {
    const key = ref.trim().toLowerCase();
    if (seenRefs.has(key)) issues.push(`Duplicate account column: "${ref}"`);
    seenRefs.add(key);
  }

  const rows: MatrixRow[] = [];
  const seenDates = new Set<string>();
  for (const line of grid.slice(1)) {
    const date = line[0] ?? "";
    if (seenDates.has(date)) issues.push(`Duplicate date: ${date}`);
    seenDates.add(date);
    const cells: MatrixCell[] = [];
    accountRefs.forEach((accountRef, i) => {
      const rawAmount = line[i + 1] ?? "";
      if (rawAmount !== "") cells.push({ accountRef, rawAmount });
    });
    rows.push({ date, cells });
  }
  return { rows, issues };
}

function parseJsonMatrix(content: string): {
  rows: MatrixRow[];
  issues: string[];
} {
  let parsed: unknown;
  try {
    parsed = JSON.parse(content);
  } catch (err) {
    return { rows: [], issues: [`Invalid JSON: ${String(err)}`] };
  }
  if (
    typeof parsed !== "object" ||
    parsed === null ||
    Array.isArray(parsed)
  ) {
    return {
      rows: [],
      issues: ["JSON root must be an object keyed by date"],
    };
  }
  const rows: MatrixRow[] = [];
  for (const [date, value] of Object.entries(parsed)) {
    if (typeof value !== "object" || value === null || Array.isArray(value)) {
      return {
        rows: [],
        issues: [`${date}: value must be an object of account → amount`],
      };
    }
    const cells: MatrixCell[] = Object.entries(value).map(
      ([accountRef, amount]) => ({ accountRef, rawAmount: String(amount) }),
    );
    rows.push({ date, cells });
  }
  return { rows, issues: [] };
}

export function buildDatePlans(
  rows: MatrixRow[],
  accounts: ApiAccount[],
  todayStr: string,
): { plans: DatePlan[]; issues: string[] } {
  const issues: string[] = [];
  const plans: DatePlan[] = [];

  for (const row of rows) {
    if (row.cells.length === 0) continue;
    if (!isDateString(row.date)) {
      issues.push(`${row.date}: invalid date`);
    } else if (row.date > todayStr) {
      issues.push(`${row.date}: future dates are not allowed`);
    }

    const snapshots: DatePlan["snapshots"] = [];
    for (const cell of row.cells) {
      const account = resolveAccountRef(cell.accountRef, accounts);
      if (!account) {
        issues.push(
          `${row.date} / "${cell.accountRef}": unknown account (${accountListHint(accounts)})`,
        );
      }
      let amountMinorRaw: number | undefined;
      try {
        amountMinorRaw = parseMajorAmount(cell.rawAmount);
      } catch (err) {
        issues.push(
          `${row.date} / "${cell.accountRef}": ${err instanceof Error ? err.message : String(err)}`,
        );
      }
      if (account && amountMinorRaw !== undefined) {
        snapshots.push({
          financialAccountId: account.id,
          accountName: account.name,
          amountMinorRaw,
        });
      }
    }
    plans.push({ date: row.date, snapshots });
  }

  if (issues.length > 0) return { plans: [], issues };
  plans.sort((a, b) => (a.date < b.date ? -1 : 1));
  return { plans, issues };
}
```

```ts
import type { Command } from "commander";
import { makeClient, fail } from "../index";
import { ApiError } from "../client";
import { parseImportFile, buildDatePlans } from "../lib/import-matrix";
import { formatMinor } from "../lib/amounts";
import { today } from "@/lib/date";

export function registerImportCommand(program: Command): void {
  program
    .command("import")
    .description("Bulk import balances from a CSV or JSON matrix file")
    .argument("<file>", "path to .csv or .json file")
    .action(async (file: string) => {
      try {
        const bunFile = Bun.file(file);
        if (!(await bunFile.exists())) {
          console.error(`File not found: ${file}`);
          process.exit(2);
        }
        const content = await bunFile.text();

        const { rows, issues: parseIssues } = parseImportFile(
          path.basename(file),
          content,
        );
        const client = await makeClient(program);
        const accounts = await client.listAccounts();
        const { plans, issues: planIssues } = buildDatePlans(
          rows,
          accounts,
          today(),
        );
        const issues = [...parseIssues, ...planIssues];
        if (issues.length > 0) {
          console.error(`${issues.length} issue(s) — nothing was written:`);
          for (const issue of issues) console.error(`  ${issue}`);
          process.exit(2);
        }
        if (plans.length === 0) {
          console.log("Nothing to import.");
          return;
        }

        const applied: string[] = [];
        for (const plan of plans) {
          try {
            await client.bulkSnapshots({
              effectiveDate: plan.date,
              snapshots: plan.snapshots.map(
                ({ financialAccountId, amountMinorRaw }) => ({
                  financialAccountId,
                  amountMinorRaw,
                }),
              ),
            });
          } catch (err) {
            console.error(
              `\nFailed at ${plan.date}: ${err instanceof ApiError ? `${err.code}: ${err.message}` : String(err)}`,
            );
            if (applied.length > 0) {
              console.error(`Applied: ${applied.join(", ")}`);
            }
            const remaining = plans
              .slice(plans.indexOf(plan))
              .map((p) => p.date);
```

```tsx
import { useEffect, useState } from "react";
import { authClient } from "@/client/auth-client";
import { Button } from "@/components/ui/button";
import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card";

type State =
  | { step: "loading" }
  | { step: "signed-out" }
  | { step: "ready" }
  | { step: "working" }
  | { step: "done" }
  | { step: "manual"; key: string }
  | { step: "error"; message: string };

export function CliAuthApp() {
  const [state, setState] = useState<State>({ step: "loading" });

  const params = new URLSearchParams(window.location.search);
  const port = params.get("port");
  const host = params.get("name") ?? "unknown-host";

  useEffect(() => {
    authClient.getSession().then(({ data }) => {
      setState(data ? { step: "ready" } : { step: "signed-out" });
    });
  }, []);

  const authorize = async () => {
    setState({ step: "working" });
    const name = `cli-${host}-${new Date().toISOString().slice(0, 10)}`;
    const { data, error } = await authClient.apiKey.create({ name });
    if (error || !data) {
      setState({
        step: "error",
        message: error?.message ?? "Failed to create API key",
      });
      return;
    }
    if (!port) {
      setState({ step: "manual", key: data.key });
      return;
    }
    try {
      const res = await fetch(`http://127.0.0.1:${port}/`, {
        method: "POST",
        headers: { "content-type": "text/plain" },
        body: data.key,
      });
      if (!res.ok) throw new Error(`loopback responded ${res.status}`);
      setState({ step: "done" });
    } catch {
      // CLI not reachable (headless/SSH) — show the key once.
      setState({ step: "manual", key: data.key });
    }
  };

  return (
    <div className="grid min-h-screen place-items-center">
      <Card className="w-full max-w-md">
        <CardHeader className="text-center">
          <CardTitle className="text-2xl">Authorize Cashy CLI</CardTitle>
          <CardDescription>
            Grants the CLI on <strong>{host}</strong> full access to your
            cashy account via an API key.
          </CardDescription>
        </CardHeader>
        <CardContent className="flex flex-col gap-4">
          {state.step === "loading" && <p>Checking session…</p>}
          {state.step === "signed-out" && (
            <>
              <p className="text-sm">
                You are not signed in. Sign in first, then return to this tab.
              </p>
              <Button asChild>
                <a href="/login" target="_blank" rel="noreferrer">
                  Go to sign in
                </a>
              </Button>
              <Button
                variant="outline"
                onClick={() => window.location.reload()}
              >
                I signed in — retry
              </Button>
            </>
          )}
          {(state.step === "ready" || state.step === "working") && (
            <Button onClick={authorize} disabled={state.step === "working"}>
              {state.step === "working" ? "Authorizing…" : "Authorize"}
            </Button>
          )}
          {state.step === "done" && (
            <p className="text-sm">
              Logged in — you can close this tab and return to the terminal.
            </p>
          )}
          {state.step === "manual" && (
            <>
              <p className="text-sm">
                Could not reach the CLI automatically. Copy this token (shown
                once) and run{" "}
                <code>bun run cashy auth login --manual</code>:
              </p>
              <pre className="overflow-x-auto rounded-md bg-muted p-3 text-xs">
                {state.key}
              </pre>
            </>
          )}
          {state.step === "error" && (
            <p className="text-sm text-destructive">{state.message}</p>
          )}
        </CardContent>
      </Card>
    </div>
  );
}
```

- [ ] **Step 1: Implement `src/cli/commands/auth.ts`**

The loopback server must answer CORS preflights: the page runs on the app origin and fetches `http://127.0.0.1:<port>` cross-origin (Chrome additionally sends a private-network preflight).

```ts
import { hostname } from "node:os";
import type { Command } from "commander";
import { makeClient, fail } from "../index";
import { CashyClient } from "../client";
import {
  CONFIG_PATH,
  DEFAULT_API_URL,
  deleteConfigFile,
  readConfigFile,
  writeConfigFile,
} from "../config";

const CORS_HEADERS = {
  "access-control-allow-origin": "*",
  "access-control-allow-methods": "POST, OPTIONS",
  "access-control-allow-headers": "content-type",
  "access-control-allow-private-network": "true",
};

function receiveToken(timeoutMs = 5 * 60 * 1000): {
  port: number;
  token: Promise<string>;
  stop: () => void;
} {
  let resolve!: (token: string) => void;
  let reject!: (err: Error) => void;
  const token = new Promise<string>((res, rej) => {
    resolve = res;
    reject = rej;
  });

  const server = Bun.serve({
    hostname: "127.0.0.1",
    port: 0,
    async fetch(req) {
      if (req.method === "OPTIONS") {
        return new Response(null, { status: 204, headers: CORS_HEADERS });
      }
      if (req.method === "POST") {
        const body = (await req.text()).trim();
        if (body.length > 0) {
          resolve(body);
          return new Response("ok", { headers: CORS_HEADERS });
        }
        return new Response("empty token", {
          status: 400,
          headers: CORS_HEADERS,
        });
      }
      return new Response("not found", { status: 404, headers: CORS_HEADERS });
    },
  });

  const timer = setTimeout(() => {
    reject(new Error("Timed out waiting for the browser (5 minutes)"));
    server.stop(true);
  }, timeoutMs);

  return {
    port: server.port,
    token: token.finally(() => {
      clearTimeout(timer);
      // Let the in-flight response flush before stopping.
      setTimeout(() => server.stop(true), 100);
    }),
    stop: () => {
      clearTimeout(timer);
      server.stop(true);
    },
  };
}

async function promptLine(question: string): Promise<string> {
  process.stdout.write(question);
  for await (const line of console) {
    return line.trim();
  }
  return "";
}

async function verifyAndSave(apiUrl: string, token: string): Promise<void> {
  const client = new CashyClient(apiUrl, token);
  const session = await client.getSession();
  await writeConfigFile({ apiUrl, token });
  console.log(
    `Logged in as ${session.user.email} (workspace: ${
      session.workspaces.find((w) => w.id === session.activeWorkspaceId)?.name ??
      session.activeWorkspaceId
    })`,
  );
  console.log(`Config written to ${CONFIG_PATH}`);
}

export function registerAuthCommands(program: Command): void {
  const auth = program.command("auth").description("Authentication");

  auth
    .command("login")
    .description("Log in via the browser (token saved to config)")
    .option("--no-open", "print the URL instead of opening the browser")
    .option("--manual", "prompt for API URL and token instead")
    .action(async (opts: { open: boolean; manual?: boolean }) => {
      try {
        const globals = program.opts<{ apiUrl?: string }>();
        const fileConfig = await readConfigFile();
        const apiUrl =
          globals.apiUrl ??
          process.env.CASHY_API_URL ??
          fileConfig.apiUrl ??
          DEFAULT_API_URL;

        if (opts.manual) {
          const enteredUrl =
            (await promptLine(`API URL [${apiUrl}]: `)) || apiUrl;
          const token = await promptLine("Token: ");
          if (!token) {
            console.error("No token entered.");
            process.exit(2);
          }
          await verifyAndSave(enteredUrl, token);
          return;
        }

        const receiver = receiveToken();
        const url = new URL("/cli/auth", apiUrl);
        url.searchParams.set("port", String(receiver.port));
        url.searchParams.set("name", hostname());

        if (opts.open) {
          console.log(`Opening ${url.href} …`);
          Bun.spawn(["open", url.href]).unref();
        } else {
          console.log(`Open this URL in your browser:\n\n  ${url.href}\n`);
        }
        console.log("Waiting for authorization (Ctrl-C to abort)…");

        const token = await receiver.token;
        await verifyAndSave(apiUrl, token);
      } catch (err) {
        fail(err);
      }
    });

  auth
    .command("whoami")
    .description("Show the logged-in user and workspaces")
    .action(async () => {
      try {
        const client = await makeClient(program);
        const session = await client.getSession();
        if (program.opts().json) {
          console.log(JSON.stringify(session, null, 2));
          return;
        }
        console.log(`${session.user.name} <${session.user.email}>`);
        for (const ws of session.workspaces) {
          const active = ws.id === session.activeWorkspaceId ? " (active)" : "";
          console.log(`  workspace: ${ws.name}${active}`);
        }
      } catch (err) {
        fail(err);
      }
    });

  auth
    .command("logout")
    .description("Delete the stored token")
    .action(async () => {
      await deleteConfigFile();
      console.log(`Removed ${CONFIG_PATH}`);
    });
}
```
